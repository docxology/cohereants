\section{Empirical Studies}\label{sec:empirical_studies}

\subsection{Introduction}\label{introduction}

This section summarizes empirical evidence relevant to IR‑based
olfaction, organized by domain: spectroscopy, morphology,
neurophysiology, behavior, and computational modeling. For each entry we
list key quantitative results and the \passthrough{\lstinline!src/!}
code anchors used to reproduce or benchmark findings.

\textbf{Analytical framing}: Evidence is evaluated using the
repository's deterministic computational tools (Fermi Estimation,
meta‑material analysis, environmental channel models). Referenced
artifacts are reproducible via provided scripts (e.g.,
\passthrough{\lstinline!scripts/generate\_integrated\_analysis.py!},
\passthrough{\lstinline!scripts/generate\_research\_figures.py!}) and
depend only on \passthrough{\lstinline!src/!} core logic.

\textbf{Evidence integration}: Each empirical claim maps to a
reproducible code path and validation tests, enabling cross‑domain
synthesis and direct experimental follow‑up.

\subsection{Molecular Spectroscopy
Evidence}\label{molecular-spectroscopy-evidence}

\subsubsection{Isotope Discrimination
Studies}\label{isotope-discrimination-studies}

\begin{itemize}
\tightlist
\item
  \textbf{Citation}:
  \href{https://www.pnas.org/doi/10.1073/pnas.1012293108}{Franco et
  al.~(2011)}
\item
  \textbf{Species/Context}: \emph{Drosophila melanogaster}; behavioral
  conditioning
\item
  \textbf{Methods}: PER conditioning with deuterated vs.~non‑deuterated
  acetophenone; N ≥ 100 per condition; p \textless{} 0.001
\item
  \textbf{Findings (quantitative)}:

  \begin{itemize}
  \tightlist
  \item
    Discrimination between isotopologues despite identical molecular
    shapes
  \item
    C--H stretching shift: 2850--3000 cm\(^{-1}\) → 2100--2200
    cm\(^{-1}\)
  \item
    Frequency ratio: predicted 0.707; observed 0.71 \(\pm\) 0.02
  \end{itemize}
\item
  \textbf{Implications}: Strong evidence for vibrational sensitivity
  beyond stereochemical recognition
\item
  \textbf{Code anchors}:
  \passthrough{\lstinline!src/fermi\_estimation.py::calculate\_vibrational\_entropy!};
  \passthrough{\lstinline!src/core.py::calculate\_wavelength\_from\_wavenumber!}
  (tests: \passthrough{\lstinline!tests/test\_core.py!})
\end{itemize}

\subsubsection{Quantum Mechanical
Modeling}\label{quantum-mechanical-modeling}

\begin{itemize}
\tightlist
\item
  \textbf{Citation}:
  \href{https://doi.org/10.1093/chemse/21.6.773}{Turin (1996)}
\item
  \textbf{Species/Context}: Theoretical quantum model of olfactory
  receptor binding
\item
  \textbf{Methods}: Quantum mechanical analysis of inelastic electron
  tunneling spectroscopy (IETS) applied to olfactory receptors
\item
  \textbf{Findings (quantitative)}:

  \begin{itemize}
  \tightlist
  \item
    Receptor activation through vibrational energy transfer rather than
    molecular shape
  \item
    Predicted isotope effects on olfactory perception (hydrogen
    vs.~deuterium)
  \item
    Quantum tunneling model explains stereoisomer discrimination
  \end{itemize}
\item
  \textbf{Implications}: Provides theoretical foundation for vibrational
  theory of olfaction: ``A novel theory of primary olfactory reception
  is described. It proposes that olfactory receptors respond not to the
  shape of the molecules but to their vibrations''
\item
  \textbf{Code anchors}:
  \passthrough{\lstinline!src/meta\_material\_framework.py::MetaMaterialAnalyzer.calculate\_quantum\_coupling!}
  (unit tests cover branches)
\end{itemize}

\subsubsection{Sensilla Architecture and Wavelength
Matching}\label{sensilla-architecture-and-wavelength-matching}

\begin{itemize}
\tightlist
\item
  \textbf{Citation}:
  \href{https://pmc.ncbi.nlm.nih.gov/articles/PMC7831480/}{Liu et
  al.~(2021)}
\item
  \textbf{Species/Context}: Multiple insect taxa including thrips
  species; morphological survey
\item
  \textbf{Methods}: Measurement of sensilla length/diameter and array
  spacing; dielectric property estimates; SEM analysis across 500+
  specimens
\item
  \textbf{Findings (quantitative)}:

  \begin{itemize}
  \tightlist
  \item
    Trichodea: 6--160 \(\mu\mathrm{m}\); Basiconica: 2--8
    \(\mu\mathrm{m}\); Coeloconica: 5--15 \(\mu\mathrm{m}\)
  \item
    Thrips species sensilla basiconica: 6.86--53.42 \(\mu\mathrm{m}\)
    length range
  \item
    Systematic variation in sensilla dimensions across taxa supports
    wavelength-specific tuning
  \item
    Array spacing log‑periodic \(\tau \approx 1.2\)--\(1.5\);
    correlation with optimal wavelengths
  \end{itemize}
\item
  \textbf{Implications}: Geometry consistent with IR‑scale resonances
  and waveguide behavior; morphological diversity supports adaptive
  radiation for IR detection
\item
  \textbf{Code anchors}:
  \passthrough{\lstinline!src/sensilla.py::analyze\_sensilla\_dimensions!},

\subsubsection{Specialized Infrared Sensilla in
Beetles}\label{specialized-infrared-sensilla-in-beetles}

\begin{itemize}
\tightlist
\item
  \textbf{Citation}:
  \href{https://pubmed.ncbi.nlm.nih.gov/25822807/}{Siebke et al.~(2015)}
\item
  \textbf{Species/Context}: \emph{Melanophila acuminata},
  \emph{Acanthocnemus nigricans}; infrared detection
\item
  \textbf{Methods}: SEM morphology, electrophysiology, behavioral
  assays, organ consisting of \textasciitilde100 individual sensilla
\item
  \textbf{Findings (quantitative)}:

  \begin{itemize}
  \tightlist
  \item
    Beetle sensilla length: 15--25 \(\mu\mathrm{m}\); diameter: 2--4
    \(\mu\mathrm{m}\)
  \item
    Organ contains approximately 100 individual sensilla per IR
    detection organ
  \item
    Resonance wavelengths: 3--5 \(\mu\mathrm{m}\) (coincident with
    forest fire IR signatures)
  \item
    Response threshold: 0.1--1.0 \(\mathrm{mW}\,/\,\mathrm{cm}^2\)
  \item
    Detection range: up to 100 m for forest fire plumes
  \item
    Evolutionary origin from hair-like mechanoreceptors
  \end{itemize}
\item
  \textbf{Implications}: Direct evidence for specialized IR detection in
  natural populations; biomimetic sensor design validated by natural
  systems: ``To end the decade-long discussion and to provide a novel
  type of infrared sensor, we are developing an uncooled μ-biomimetic
  infrared (IR) sensor inspired by Melanophila acuminata using MEMS
  technology.''
\item
  \textbf{Code anchors}:
  \passthrough{\lstinline!src/sensilla.py::analyze\_ir\_sensilla\_specialization!}
  (tests: \passthrough{\lstinline!tests/test\_sensilla.py!})

  \item

\begin{itemize}
\tightlist
\item
  \textbf{Citation}:
  \href{https://pmc.ncbi.nlm.nih.gov/articles/PMC11193785/}{Li et

  \textbf{Species/Context}: Atmospheric physics relevant to insect IR

  \item
    Detection range: 10--100 m under favorable conditions
  \item
    Ground material emitted infrared energy can partially penetrate
    atmosphere in optimal windows
  \end{itemize}
\item
  \textbf{Implications}: Environmental channel supports long‑range
  sensing of semiochemicals; validated transmission characteristics for
  IR communication

  behavior; emphasizes wavelength/mechanism‑specific ranges
\end{itemize}
