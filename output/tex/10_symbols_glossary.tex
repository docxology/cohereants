\section{Symbols and Glossary}\label{sec:symbols_glossary}

\subsection{Key Terms and Definitions}\label{key-terms-and-definitions}

\subsubsection{Olfaction and
Chemosensation}\label{olfaction-and-chemosensation}

\begin{itemize}
\tightlist
\item
  \textbf{Olfaction}: The sense of smell; the ability to detect and
  identify airborne molecules through specialized sensory organs
\item
  \textbf{Chemosensation}: The detection of chemical stimuli by sensory
  cells, including olfaction, gustation, and chemesthesis
\item
  \textbf{Semiochemicals}: Chemical substances that carry information
  between organisms, including pheromones, allomones, and kairomones
\item
  \textbf{Pheromones}: Semiochemicals that affect the behavior of other
  members of the same species, such as sex pheromones and trail
  pheromones
\item
  \textbf{Cuticular Hydrocarbons (CHCs)}: Long-chain hydrocarbons found
  on the surface of insects that serve as recognition cues and
  waterproofing agents
\item
  \textbf{Sensilla}: Microscopic sensory hairs or pegs on insect
  antennae and other body parts that serve as the primary sensory units
  for olfaction and other senses
\end{itemize}

\subsubsection{Insect Anatomy and
Physiology}\label{insect-anatomy-and-physiology}

\begin{itemize}
\tightlist
\item
  \textbf{Antennae}: Paired sensory appendages on the head of insects
  that contain olfactory and other sensory receptors
\item
  \textbf{Sensilla}: Microscopic sensory hairs or pegs on insect
  antennae and other body parts that serve as the primary sensory units
\item
  \textbf{Sensilla Trichodea}: Hair-like sensilla that are often
  involved in olfaction, typically 6-160 μm in length
\item
  \textbf{Sensilla Basiconica}: Peg-like sensilla with porous surfaces,
  typically 2-8 μm in length
\item
  \textbf{Sensilla Coeloconica}: Pit-like sensilla that may detect
  temperature, humidity, and infrared radiation
\item
  \textbf{ORN}: Olfactory Receptor Neuron; nerve cells that respond to
  chemical stimuli and transmit signals to the brain
\item
  \textbf{OR}: Olfactory Receptor; membrane proteins that bind to odor
  molecules and initiate signal transduction
\item
  \textbf{Antennal Lobe (AL)}: First olfactory processing center in the
  insect brain containing glomeruli that aggregate ORN inputs by
  receptor type
\item
  \textbf{Glomerulus (plural: glomeruli)}: Spheroidal neuropil
  compartment in the AL where ORN axons synapse with projection neurons
  and local interneurons; often tuned to receptor families or
  vibrational features
\end{itemize}

\subsubsection{Electromagnetic Theory and Infrared
Detection}\label{electromagnetic-theory-and-infrared-detection}

\begin{itemize}
\tightlist
\item
  \textbf{Infrared (IR)}: Electromagnetic radiation with wavelengths
  longer than visible light (0.7-1000 μm), invisible to human eyes but
  detectable by specialized sensors
\item
  \textbf{Mid-infrared (MIR)}: IR radiation in the 2-25 μm range,
  corresponding to molecular vibrational modes and fundamental for
  chemical sensing applications
\item
  \textbf{Far-infrared (FIR)}: IR radiation in the 25-1000 μm range,
  corresponding to rotational and low-frequency vibrational modes, also
  known as thermal infrared
\item
  \textbf{Near-infrared (NIR)}: IR radiation in the 0.7-2 μm range, just
  beyond visible light, commonly used in spectroscopy and optical
  communications
\item
  \textbf{Dielectric}: A material that can be polarized by an electric
  field and supports electromagnetic wave propagation
\item
  \textbf{Waveguide}: A structure that guides electromagnetic waves
  along a specific path with minimal loss
\item
  \textbf{Resonator}: A device or structure that oscillates at specific
  frequencies, amplifying signals at resonant frequencies
\item
  \textbf{Quality Factor (Q)}: A measure of resonator performance,
  defined as the ratio of stored energy to energy lost per cycle
\end{itemize}

\subsubsection{Spectroscopy and Molecular
Properties}\label{spectroscopy-and-molecular-properties}

\begin{itemize}
\tightlist
\item
  \textbf{Vibrational Theory}: The hypothesis that olfaction works by
  detecting molecular vibrations in the infrared spectrum rather than
  molecular shape, providing a mechanism for odor recognition at the
  quantum mechanical level
\item
  \textbf{Emission Spectrum}: The range of wavelengths of
  electromagnetic radiation emitted by a substance when excited,
  characteristic of the energy level transitions in the material
\item
  \textbf{Absorption Spectrum}: The range of wavelengths absorbed by a
  substance, complementary to emission spectra and determined by the
  molecular structure and bonding
\item
  \textbf{Transmission Window}: A range of wavelengths where the
  atmosphere is relatively transparent to electromagnetic radiation,
  allowing for long-range signal propagation
\item
  \textbf{Deuteration}: The replacement of hydrogen atoms with deuterium
  (heavy hydrogen) in molecules, affecting vibrational frequencies
\item
  \textbf{Enantiomers}: Mirror-image forms of the same molecule that may
  have different olfactory properties
\item
  \textbf{FRET}: Förster Resonance Energy Transfer; energy transfer
  between molecules through dipole-dipole interactions
\item
  \textbf{Wavenumber}: The reciprocal of wavelength, typically expressed
  in cm\(^{-1}\), related to energy by \(E = hc\tilde{\nu}\)
\end{itemize}

\subsection{Mathematical Notation}\label{mathematical-notation}

\subsubsection{Wavelength and frequency}\label{wavelength-and-frequency}

\begin{itemize}
\tightlist
\item
  \textbf{λ (lambda)}: Wavelength, typically in micrometers (μm) or
  nanometers (nm).
\item
  \textbf{ν (nu)}: Frequency in Hz, related to wavelength by
  \(c = \lambda\nu\).
\item
  \textbf{\(\tilde{\nu}\) (wavenumber)}: Reciprocal wavelength in
  cm\(^{-1}\), \(\tilde{\nu} = 10^4/\lambda\) (for λ in μm).
\item
  \textbf{c}: Speed of light in vacuum (2.998 × 10\^{}8 m/s).
\item
  \textbf{μm}: Micrometer (10\^{}-6 m); standard unit for infrared
  wavelengths.
\item
  \textbf{nm}: Nanometer (10\^{}-9 m).
\item
  \textbf{cm\^{}-1}: Wavenumber unit used in IR spectroscopy.
\end{itemize}

\subsubsection{Atmospheric Transmission
Windows}\label{atmospheric-transmission-windows}

The Earth's atmosphere has specific wavelength ranges where infrared
radiation can travel relatively freely, enabling long-range detection of
insect semiochemicals:

\begin{itemize}
\tightlist
\item
  \textbf{2-5 μm (Mid-infrared)}: \textasciitilde80\% transmission
  efficiency, optimal for hydrocarbon detection
\item
  \textbf{8-14 μm (Long-wave infrared)}: \textasciitilde90\%
  transmission efficiency, optimal for long-range communication; ground
  materials can emit infrared energy that partially penetrates this
  window
\item
  \textbf{17-25 μm (Far-infrared)}: \textasciitilde70\% transmission
  efficiency, useful for thermal detection
\end{itemize}

\textbf{Transmission function}: Modeled by
\passthrough{\lstinline!src/core.py::calculate\_atmospheric\_transmission()!}
with validation against peer-reviewed atmospheric physics studies (see
\eqref{eq:atmospheric_transmission}; unit tests in
\passthrough{\lstinline!tests/test\_core.py!}):

\begin{equation}
T(\lambda) = \exp\left[-\sum_i \alpha_i(\lambda) L_i\right]
\label{eq:transmission_function_gloss}
\end{equation}

where \(\alpha_i(\lambda)\) is the absorption coefficient and \(L_i\) is
the path length through atmospheric component \(i\).

\subsubsection{Sensilla Dimensions and Wavelength
Matching}\label{sensilla-dimensions-and-wavelength-matching}

Insect sensilla have dimensions that correspond closely to the
wavelengths of infrared radiation they detect, as confirmed by
comparative morphometric studies:

\begin{itemize}
\tightlist
\item
  \textbf{Sensilla Trichodea}: 6-160 μm length, optimal for 2-30 μm
  wavelengths
\item
  \textbf{Sensilla Basiconica}: 2-8 μm length, optimal for 1-10 μm
  wavelengths; specific dimensions of 6.86--53.42 μm observed in thrips
  species
\item
  \textbf{Sensilla Coeloconica}: 5-15 μm length, optimal for 3-20 μm
  wavelengths
\item
  \textbf{Specialized IR organs}: Approximately 100 sensilla per organ
  in beetle species
\end{itemize}

\textbf{Wavelength matching}: Analyzed by
\passthrough{\lstinline!src/sensilla.py::analyze\_sensilla\_dimensions()!}
with validation against peer-reviewed morphometric data; see resonant
frequency \eqref{eq:resonant_freq_gloss} and tests
\passthrough{\lstinline!tests/test\_sensilla.py!}. Publication figures
are generated via
\passthrough{\lstinline!scripts/generate\_research\_figures.py!}.

\textbf{Resonant Frequency}: The fundamental resonant frequency of a
sensillum is:

\begin{equation}
f_{res} = \frac{c}{2\pi} \sqrt{\left(\frac{\alpha_{mn}}{a}\right)^2 + \left(\frac{p\pi}{L}\right)^2}
\label{eq:resonant_freq_gloss}
\end{equation}

where \(c\) is the speed of light, \(\alpha_{mn}\) is the Bessel
function root, and \(a\) and \(L\) are the radius and length.

\subsubsection{Signal Processing and Information
Theory}\label{signal-processing-and-information-theory}

The vibrational theory incorporates advanced signal processing concepts:

\begin{itemize}
\tightlist
\item
  \textbf{Channel Capacity}: The maximum information rate that can be
  transmitted through the infrared detection channel:
\end{itemize}

\begin{equation}
C = B \log_2(1 + SNR)
\label{eq:channel_capacity_gloss}
\end{equation}

where \(B\) is the bandwidth and \(SNR\) is the signal-to-noise ratio.

\begin{itemize}
\tightlist
\item
  \textbf{Detection Threshold}: The minimum detectable power is:
\end{itemize}

\begin{equation}
P_{min} = k_B T \Delta f \cdot SNR_{min}
\label{eq:min_power_gloss}
\end{equation}

where \(k_B\) is Boltzmann's constant, \(T\) is temperature,
\(\Delta f\) is bandwidth, and \(SNR_{min}\) is the minimum required
signal-to-noise ratio.

\subsection{Research Methodology
Terms}\label{research-methodology-terms}

\subsubsection{Experimental Techniques}\label{experimental-techniques}

\begin{itemize}
\tightlist
\item
  \textbf{Ionotropic}: Direct ligand-gated ion channels that open
  immediately upon binding
\item
  \textbf{Metabotropic}: G-protein coupled receptor systems that
  activate intracellular signaling cascades
\item
  \textbf{Behavioral Conditioning}: Training insects to associate
  specific stimuli with rewards or punishments
\item
  \textbf{Electroantennography (EAG)}: Recording electrical responses
  from insect antennae to chemical stimuli
\item
  \textbf{Single Sensillum Recording}: Recording from individual
  sensilla to measure response characteristics
\end{itemize}

\subsubsection{Physical and Chemical
Properties}\label{physical-and-chemical-properties}

\begin{itemize}
\tightlist
\item
  \textbf{Piezoelectric}: Materials that generate electric charge in
  response to mechanical stress
\item
  \textbf{Allosteric Modulation}: Changes in protein function due to
  binding at sites other than the active site
\item
  \textbf{Photomodulation}: Changes in protein function due to light
  absorption
\item
  \textbf{Dielectric Loss}: Energy dissipation in dielectric materials
  due to molecular motion
\item
  \textbf{Resonant Coupling}: Enhanced energy transfer when systems
  oscillate at the same frequency
\end{itemize}

\subsubsection{Statistical and Analytical
Methods}\label{statistical-and-analytical-methods}

\begin{itemize}
\tightlist
\item
  \textbf{Power Analysis}: Statistical method to determine the minimum
  sample size needed to detect an effect
\item
  \textbf{Receiver Operating Characteristic (ROC)}: Plot of true
  positive rate vs.~false positive rate
\item
  \textbf{Discriminability Index (d')}: Measure of ability to
  distinguish between signal and noise
\item
  \textbf{Hill Coefficient}: Measure of cooperativity in binding or
  response functions
\item
  \textbf{Log-Periodic Analysis}: Analysis of systems with periodic
  spacing that increases logarithmically
\end{itemize}
